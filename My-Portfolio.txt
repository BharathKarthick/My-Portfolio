import jwt from "jsonwebtoken";

export function requireAdmin(req, res, next) {
  const authorization = req.headers.authorization;

  if (!authorization?.startsWith("Bearer ")) {
    return res.status(401).json({
      success: false,
      message: "Authentication required",
    });
  }

  const token = authorization.slice(7);

  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);

    if (decoded.role !== "admin") {
      return res.status(403).json({
        success: false,
        message: "Admin access required",
      });
    }

    req.admin = {
      username: decoded.username,
      role: decoded.role,
    };

    next();
  } catch {
    return res.status(401).json({
      success: false,
      message: "Invalid or expired token",
    });
  }
}

router.put("/", async (req, res) => {
  try {
    const { name, title, image, paragraphs, infoCards } = req.body ?? {};

    let about = await AboutModel.findOne();

    const payload = {
      name: name || "P. Bharath",
      title: title || "Full Stack Developer",
      image: image || about?.image || "",
      paragraphs: Array.isArray(paragraphs) ? paragraphs : [],
      infoCards: Array.isArray(infoCards) ? infoCards : [],
    };

    if (!about) {
      about = await AboutModel.create(payload);
    } else {
      about = await AboutModel.findOneAndUpdate({}, { $set: payload }, { new: true, runValidators: true });
    }

    res.json({ success: true, about });
  } catch (error) {
    res.status(500).json({ success: false, message: error.message || "Unable to update about data" });
  }
});

export default router;

import express from "express";
import mongoose from "mongoose";
import ExperienceModel from "../models/Experience.js";
import { requireAdmin } from "../middleware/auth.js";

const router = express.Router();
router.use(requireAdmin);

router.get("/", async (_req, res) => {
  try {
    const experiences = await ExperienceModel.find().sort({ order: 1, createdAt: -1 }).lean();
    res.json({ success: true, count: experiences.length, experiences });
  } catch (error) {
    res.status(500).json({ success: false, message: "Unable to fetch experiences" });
  }
});

router.post("/", async (req, res) => {
  try {
    const { company, period, role, ...rest } = req.body ?? {};
    if (!company?.trim() || !period?.trim() || !role?.trim()) {
      return res.status(400).json({ success: false, message: "Company, period and role are required" });
    }

    const experience = await ExperienceModel.create({
      company: company.trim(),
      period: period.trim(),
      role: role.trim(),
      ...rest,
    });

    res.status(201).json({ success: true, experience });
  } catch (error) {
    res.status(500).json({ success: false, message: error.message || "Unable to create experience" });
  }
});

router.put("/:id", async (req, res) => {
  try {
    if (!mongoose.isValidObjectId(req.params.id)) {
      return res.status(400).json({ success: false, message: "Invalid experience ID" });
    }

    const experience = await ExperienceModel.findByIdAndUpdate(
      req.params.id,
      { $set: req.body },
      { returnDocument: "after", runValidators: true }
    );

    if (!experience) return res.status(404).json({ success: false, message: "Experience not found" });
    res.json({ success: true, experience });
  } catch (error) {
    res.status(500).json({ success: false, message: error.message || "Unable to update experience" });
  }
});

router.delete("/:id", async (req, res) => {
  try {
    if (!mongoose.isValidObjectId(req.params.id)) {
      return res.status(400).json({ success: false, message: "Invalid experience ID" });
    }

    const experience = await ExperienceModel.findByIdAndDelete(req.params.id);
    if (!experience) return res.status(404).json({ success: false, message: "Experience not found" });
    res.json({ success: true, message: "Experience deleted successfully" });
  } catch (error) {
    res.status(500).json({ success: false, message: "Unable to delete experience" });
  }
});

export default router;

import express from "express";
import bcrypt from "bcryptjs";
import jwt from "jsonwebtoken";
import { rateLimit } from "express-rate-limit";

const router = express.Router();

const loginLimiter = rateLimit({
  windowMs: 15 * 60 * 1000, // 15 minutes
  limit: 5, // Maximum 5 attempts per IP per window
  standardHeaders: "draft-8",
  legacyHeaders: false,
  message: {
    success: false,
    message: "Too many login attempts. Please try again in 15 minutes.",
  },
});

router.post("/login", loginLimiter, async (req, res) => {
  try {
    const { username, password } = req.body ?? {};

    if (
      typeof username !== "string" ||
      typeof password !== "string" ||
      !username.trim() ||
      !password
    ) {
      return res.status(400).json({
        success: false,
        message: "Username and password are required",
      });
    }

    if (!process.env.ADMIN_PASSWORD_HASH || !process.env.JWT_SECRET) {
      return res.status(500).json({
        success: false,
        message: "Admin authentication is not configured",
      });
    }

    const usernameMatches = username === process.env.ADMIN_USERNAME;
    const passwordMatches = await bcrypt.compare(
      password,
      process.env.ADMIN_PASSWORD_HASH
    );

    if (!usernameMatches || !passwordMatches) {
      return res.status(401).json({
        success: false,
        message: "Invalid username or password",
      });
    }

    const token = jwt.sign(
      {
        username: process.env.ADMIN_USERNAME,
        role: "admin",
      },
      process.env.JWT_SECRET,
      { expiresIn: "2h" }
    );

    res.json({
      success: true,
      message: "Login successful",
      token,
      expiresIn: 7200,
    });
  } catch (error) {
    console.error("Admin login failed:", error.message);
    res.status(500).json({
      success: false,
      message: "Unable to log in",
    });
  }
});

export default router;

import { requireAdmin } from "../middleware/auth.js";

const router = express.Router();

const MAX_IMAGE_SIZE = 5 * 1024 * 1024;
const MAX_PDF_SIZE = 10 * 1024 * 1024;

const upload = multer({
  storage: multer.memoryStorage(),
  limits: {
    fileSize: MAX_PDF_SIZE,
    files: 1,
  },
  fileFilter: (_req, file, callback) => {
    const allowedTypes = ["image/jpeg", "image/png", "image/webp"];

    if (
      !allowedTypes.includes(file.mimetype) &&
      file.mimetype !== "application/pdf"
    ) {
      return callback(
        new Error("Only JPG, PNG, WebP images and PDF files are allowed")
      );
    }

    callback(null, true);
  },
});

function uploadToCloudinary(buffer, options) {
  return new Promise((resolve, reject) => {
    const stream = cloudinary.uploader.upload_stream(
      options,
      (error, result) => {
        if (error) return reject(error);
        resolve(result);
      }
    );

    stream.end(buffer);
  });
}

router.use(requireAdmin);

// 1. Project Image Upload API
router.post("/project-image", (req, res) => {
  upload.single("file")(req, res, async (error) => {
    if (error) {
      return res.status(400).json({
        success: false,
        message: error.message || "Invalid upload",
      });
    }

    if (!req.file) {
      return res.status(400).json({
        success: false,
        message: "Select an image to upload",
      });
    }

    if (req.file.size > MAX_IMAGE_SIZE) {
      return res.status(400).json({
        success: false,
        message: "Images must be 5 MB or smaller",
      });
    }

    try {
      const result = await uploadToCloudinary(req.file.buffer, {
        resource_type: "image",
        folder: "portfolio/projects",
        public_id: randomUUID(),
        overwrite: false,
      });

      res.status(201).json({
        success: true,
        message: "Project image uploaded successfully",
        url: result.secure_url,

  const parts = text.split(/(\*[^*]+\*)/g);
  return parts.map((part, i) => {

      return (
        <span key={i} className="text-[#FF5C00] font-semibold">
          {content}
        </span>
      );
    }
    return part;
  });
}

import { useEffect, useState } from "react";
import Login from "./admin/Login";
import Dashboard from "./admin/Dashboard";

const API_URL = import.meta.env.VITE_API_BASE_URL || "http://localhost:5000";
const TOKEN_KEY = "portfolio_admin_token";

export default function Admin() {
  const [checkingSession, setCheckingSession] = useState(true);
  const [token, setToken] = useState("");
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  useEffect(() => {
    let cancelled = false;

    async function checkSession() {
      const savedToken = sessionStorage.getItem(TOKEN_KEY);
      if (!savedToken) {
        if (!cancelled) setCheckingSession(false);
        return;
      }

      try {
        const response = await fetch(`${API_URL}/api/settings/home`, {
          headers: { Authorization: `Bearer ${savedToken}` },
        });

        if (!response.ok) {
          sessionStorage.removeItem(TOKEN_KEY);
          if (!cancelled) setToken("");
        } else if (!cancelled) {
          setToken(savedToken);
        }
      } catch {
        if (!cancelled) {
          setError("Unable to verify your session.");
        }
      } finally {
        if (!cancelled) setCheckingSession(false);
      }
    }

    checkSession();
    return () => {
      cancelled = true;
    };
  }, []);

  async function handleLogin({ username, password }) {
    setError("");
    setLoading(true);

    try {
      const response = await fetch(`${API_URL}/api/auth/login`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ username, password }),
      });

      const data = await response.json();
      if (!response.ok || !data.token) {
        throw new Error(data.message || "Invalid username or password.");
      }

      sessionStorage.setItem(TOKEN_KEY, data.token);
      setToken(data.token);
    } catch (err) {
      setError(err.message || "Unable to log in.");
    } finally {
      setLoading(false);
    }
  }

  function handleLogout() {
    sessionStorage.removeItem(TOKEN_KEY);
    setToken("");
    setError("");
  }

  if (checkingSession) {
    return (
      <main className="flex min-h-screen items-center justify-center bg-black font-['Plus_Jakarta_Sans'] text-white">
        <p className="text-sm text-white/50">Checking admin session...</p>
      </main>
    );
  }

  if (token) {
    return <Dashboard token={token} onLogout={handleLogout} />;
  }

  return <Login onLogin={handleLogin} loading={loading} error={error} />;
}

import { useEffect, useState, useCallback } from "react";
import PropTypes from "prop-types";
import ConfirmationModal from "../../components/ConfirmationModal";
import { Plus, Edit, Trash2, Briefcase } from "lucide-react";

const API_URL = import.meta.env.VITE_API_BASE_URL || "http://localhost:5000";

function emptyExperienceForm(initialOrder = 1) {
  return {
    company: "",
    role: "",
    period: "",
    type: "Intern",
    skills: "",
    highlights: "",
    order: initialOrder,
  };
}

function ExperienceForm({ token, experience, initialOrder, onCancel, onSaved }) {
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");
  const [form, setForm] = useState(() => {
    if (!experience) return emptyExperienceForm(initialOrder);
    return {
      company: experience.company || "",
      role: experience.role || "",
      period: experience.period || "",
      type: experience.type || "Intern",
      skills: Array.isArray(experience.skills) ? experience.skills.join(", ") : "",
      highlights: Array.isArray(experience.highlights) ? experience.highlights.join("\n") : "",
      order: experience.order ?? 1,
    };
  });

  function updateField(event) {
    const { name, value } = event.target;
    setForm((current) => ({ ...current, [name]: value }));
  }

  const initialSkills = Array.isArray(experience?.skills)
    ? experience.skills.map((s) => s.trim()).filter(Boolean).join(", ")
    : "";
  const currentSkills = form.skills.split(",").map((s) => s.trim()).filter(Boolean).join(", ");

  const initialHighlights = Array.isArray(experience?.highlights)
    ? experience.highlights.map((h) => h.trim()).filter(Boolean).join("\n")
    : "";
  const currentHighlights = form.highlights.split("\n").map((h) => h.trim()).filter(Boolean).join("\n");

  const hasChanges =
    !experience ||
    form.company.trim() !== (experience.company || "").trim() ||
    form.role.trim() !== (experience.role || "").trim() ||
    form.period.trim() !== (experience.period || "").trim() ||
    form.type.trim() !== (experience.type || "Intern").trim() ||
    currentSkills !== initialSkills ||
    currentHighlights !== initialHighlights ||
    Number(form.order) !== Number(experience.order ?? 1);

  async function handleSubmit(event) {
    event.preventDefault();
    setError("");

    if (experience && !hasChanges) {
      return;
    }

    setSaving(true);

    try {
      const payload = {
        company: form.company.trim(),
        role: form.role.trim(),
        period: form.period.trim(),
        type: form.type.trim(),
        skills: form.skills.split(",").map((s) => s.trim()).filter(Boolean),
        highlights: form.highlights.split("\n").map((h) => h.trim()).filter(Boolean),
        order: Number(form.order),
      };

      const response = await fetch(
        experience
          ? `${API_URL}/api/admin/experiences/${experience._id}`
          : `${API_URL}/api/admin/experiences`,
        {
          method: experience ? "PUT" : "POST",
          headers: {
            "Content-Type": "application/json",
            Authorization: `Bearer ${token}`,
          },
          body: JSON.stringify(payload),
        }
      );

      const data = await response.json();
      if (!response.ok || !data.success) {
        throw new Error(data.message || "Unable to save experience.");
      }
      onSaved(experience ? "updated" : "created", payload.company);
    } catch (submitError) {
      setError(submitError.message || "Unable to save experience.");
    } finally {
      setSaving(false);
    }
  }

  return (
    <form onSubmit={handleSubmit} className="space-y-6 text-white font-['Plus_Jakarta_Sans']">
      <div className="flex items-center justify-between">
        <h3 className="text-xl font-bold">{experience ? "Edit Experience" : "Add New Experience"}</h3>
        <button
          type="button"
          onClick={onCancel}
          className="rounded-xl border border-white/10 px-4 py-2 text-xs font-semibold text-white/75 hover:bg-white/5 cursor-pointer"
        >
          Cancel

    setLoading(true);
    setError("");

    try {
      const response = await fetch(`${API_URL}/api/settings/home`);
      const data = await response.json();

      if (!response.ok || !data.success) {
        throw new Error(data.message || "Unable to load Home settings.");
      }

      const home = data.home || {};
      const loaded = {
        roleTitles:
          Array.isArray(home.roleTitles) && home.roleTitles.length > 0
            ? home.roleTitles
            : DEFAULT_HOME.roleTitles,
        location: home.location ?? DEFAULT_HOME.location,
        experienceYears: Number(home.experienceYears ?? DEFAULT_HOME.experienceYears),
        terminalName: home.terminalName ?? DEFAULT_HOME.terminalName,
      };

      setForm(loaded);
      setInitialState(JSON.stringify(loaded));
    } catch (err) {
      setError(err.message || "Unable to load Home settings.");
    } finally {
      setLoading(false);
    }
  }, []);

    loadHomeSettings();
  }, [loadHomeSettings]);

  function updateField(event) {
    const { name, value } = event.target;
    setForm((current) => ({
      ...current,
      [name]: value,
    }));
  }

  function updateRole(index, value) {
    setForm((current) => ({
      ...current,
      roleTitles: current.roleTitles.map((role, roleIndex) =>
        roleIndex === index ? value : role,
      ),
    }));
  }

  function addRole() {
    setForm((current) => ({
      ...current,
      roleTitles: [...current.roleTitles, ""],
    }));
  }

  function removeRole(index) {
    setForm((current) => ({
      ...current,
      roleTitles: current.roleTitles.filter(
        (_, roleIndex) => roleIndex !== index,
      ),
    }));
    setError("");
  }

  const currentComparable = JSON.stringify({
    roleTitles: form.roleTitles.map((r) => r.trim()),
    location: form.location.trim(),
    experienceYears: Number(form.experienceYears),
    terminalName: form.terminalName.trim(),
  });

  const hasChanges = currentComparable !== initialState;

  async function handleSubmit(event) {
    event.preventDefault();
    setError("");

    // எந்த மாற்றமும் செய்யாமல் Save கிளிக் செய்தால் எதுவும் நடக்காது
    if (!hasChanges) {
      return;
    }

    setSaving(true);

    const roleTitles = form.roleTitles.map((role) => role.trim());

          "Content-Type": "application/json",

        body: JSON.stringify(payload),
      });

        throw new Error(data.message || "Unable to save Home settings.");

      setInitialState(JSON.stringify(payload));

        const response = await fetch(`${API_URL}/api/settings/resume`);
        const data = await response.json();

          throw new Error(data.message || "Unable to load resume details.");
        }

        if (!cancelled) {
          setResumeUrl(data.resumeUrl || "");
        }
      } catch (err) {
        if (!cancelled) {
          setError(err.message || "Unable to load resume details.");
        }
      } finally {
        if (!cancelled) {
          setLoading(false);
        }
      }

    return () => {
      cancelled = true;

      !selectedFile.name.toLowerCase().endsWith(".pdf")

      setError("Please select a valid PDF file.");
      return;

      return;

        title: "Resume Updated!",

import PropTypes from "prop-types";
import ConfirmationModal from "../../components/ConfirmationModal";
import { Plus, Edit, Trash2, Code2, ArrowUp, ArrowDown } from "lucide-react";

const API_URL = import.meta.env.VITE_API_BASE_URL || "http://localhost:5000";

const LEGACY_ICON_MAP = {
  FaReact: "simple-icons:react", FaNodeJs: "simple-icons:nodedotjs",
  FaHtml5: "simple-icons:html5", FaCss3Alt: "simple-icons:css3", FaGithub: "simple-icons:github",
  SiJavascript: "simple-icons:javascript", SiTypescript: "simple-icons:typescript",
  SiTailwindcss: "simple-icons:tailwindcss", SiExpress: "simple-icons:express",
  SiMongodb: "simple-icons:mongodb", SiPostman: "simple-icons:postman",
  SiNetlify: "simple-icons:netlify", SiRender: "simple-icons:render", SiAmazons3: "simple-icons:amazonaws",
};

const SIMPLE_ICON_ALIASES = { 
  html: "html5", css: "css3", js: "javascript", tailwind: "tailwindcss", 
  node: "nodedotjs", nodejs: "nodedotjs", expressjs: "express", postgres: "postgresql", 
  aws: "amazonaws", amazonaws: "amazonaws", python: "python", mysql: "mysql", 
  react: "react", javascript: "javascript", typescript: "typescript", mongodb: "mongodb", 
  docker: "docker", git: "git", github: "github", java: "java", php: "php", 
  firebase: "firebase", figma: "figma", wordpress: "wordpress", canva: "canva",
};

function normalizeIconIdentifier(icon) {
  const raw = String(icon || "").trim();
  if (!raw) return "";
  if (LEGACY_ICON_MAP[raw]) return LEGACY_ICON_MAP[raw];
  if (raw.includes(":")) return raw.toLowerCase();
  const slug = raw.toLowerCase().replace(/\s+/g, "-");
  return `simple-icons:${SIMPLE_ICON_ALIASES[slug] || slug}`;
}

function iconUrl(icon) {
  const id = normalizeIconIdentifier(icon);
  const match = id.match(/^([a-z0-9-]+):([a-z0-9-]+)$/i);
  if (!match) return null;
  return `https://api.iconify.design/${match[1]}/${match[2]}.svg`;
}

function IconPreview({ icon, color, size = 30 }) {
  const src = iconUrl(icon);
  const [failedSrc, setFailedSrc] = useState("");
  const failed = Boolean(src && failedSrc === src);

  if (!src || failed) {
    return (
      <span
        className="inline-flex shrink-0 items-center justify-center rounded-lg bg-white/5 font-bold text-slate-400"
        style={{ width: size, height: size }}
      >
        {String(icon || "?").trim().slice(0, 1).toUpperCase() || "?"}
      </span>
    );
  }

  return (
    <span
      key={src}
      className="inline-block shrink-0"
      role="img"
      aria-label="Icon preview"
      style={{
        width: size,
        height: size,
        backgroundColor: color || "#FF5C00",
        WebkitMask: `url("${src}") center / contain no-repeat`,
        mask: `url("${src}") center / contain no-repeat`,
      }}
    >
      <img src={src} alt="" width={size} height={size} onError={() => setFailedSrc(src)} style={{ display: "none" }} />
    </span>
  );
}

IconPreview.propTypes = {
  icon: PropTypes.string,
  color: PropTypes.string,
  size: PropTypes.number,
};

const emptyForm = {
  name: "",
  icon: "",
  color: "#FF5C00",
  order: 1,
  published: true,
};

export default function SkillManagement({ token }) {
  const [skills, setSkills] = useState([]);
  const [form, setForm] = useState(emptyForm);
  const [editingId, setEditingId] = useState(null);
  const [showForm, setShowForm] = useState(false);
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [deletingId, setDeletingId] = useState("");
  const [error, setError] = useState("");
  const [pendingDelete, setPendingDelete] = useState(null);
  const [modal, setModal] = useState(null);

  const loadSkills = useCallback(async () => {
    setLoading(true);
    setError("");

    try {
      const response = await fetch(`${API_URL}/api/admin/skills`, {
        headers: {
          Authorization: `Bearer ${token}`,
        },
      });

      const data = await response.json();

      if (!response.ok || !data.success) {
        throw new Error(data.message || "Unable to load skills.");
      }

      setSkills(Array.isArray(data.skills) ? data.skills : []);
    } catch (err) {
      setError(err.message || "Unable to load skills.");
    } finally {
      setLoading(false);
    }
  }, [token]);

  useEffect(() => {
    loadSkills();
  }, [loadSkills]);

  function startAdd() {
    const nextOrder =
      skills.length > 0
        ? Math.max(...skills.map((skill) => Number(skill.order) || 0)) + 1
        : 1;

    setEditingId(null);
    setForm({
      ...emptyForm,
      order: nextOrder,
    });
    setError("");
    setShowForm(true);
  }

  function startEdit(skill) {
    setEditingId(skill._id);
    setForm({
      name: skill.name || "",
      icon: skill.icon || skill.key || "",
      color: skill.color || "#FF5C00",
      order: skill.order ?? 1,
      published: skill.published ?? true,
    });
    setError("");
    setShowForm(true);
  }

  function updateField(event) {
    const { name, value, type, checked } = event.target;

    setForm((current) => ({
      ...current,
      [name]: type === "checkbox" ? checked : value,
    }));
  }

  async function handleSubmit(event) {
    event.preventDefault();
    setSaving(true);
    setError("");

    try {
      if (!form.name.trim()) throw new Error("Please enter a skill name.");
      if (!form.icon.trim()) throw new Error("Please enter an icon name or Iconify identifier.");
      if (!/^#(?:[0-9a-f]{3}|[0-9a-f]{6})$/i.test(form.color)) throw new Error("Please choose a valid hexadecimal color.");
      
      const payload = {
        name: form.name.trim(),
        icon: normalizeIconIdentifier(form.icon),
        key: normalizeIconIdentifier(form.icon), // compatibility with existing schema
        color: form.color,
        order: Number(form.order),
        published: form.published,
      };

      const response = await fetch(
        editingId
          ? `${API_URL}/api/admin/skills/${editingId}`
          : `${API_URL}/api/admin/skills`,
        {
          method: editingId ? "PUT" : "POST",
          headers: {
            "Content-Type": "application/json",
            Authorization: `Bearer ${token}`,
          },
          body: JSON.stringify(payload),
        },
      );

      const data = await response.json();

      if (!response.ok || !data.success) {
        throw new Error(data.message || "Unable to save skill.");
      }

      const wasEditing = Boolean(editingId);
      const skillName = payload.name;

      setShowForm(false);
      setEditingId(null);
      setForm({ ...emptyForm });

      await loadSkills();

      setModal({
        type: "success",
        variant: "success",
        title: wasEditing ? "Skill Updated!" : "Skill Added!",
        message: wasEditing
          ? `"${skillName}" was updated successfully.`
          : `"${skillName}" was added successfully.`,
        confirmText: "Done",
        cancelText: "Close",
      });
    } catch (err) {
      setError(err.message || "Unable to save skill.");
    } finally {
      setSaving(false);
    }
  }

  async function confirmDelete() {
    if (!pendingDelete) return;

    const skill = pendingDelete;
    setDeletingId(skill._id);
    setError("");

    try {
      const response = await fetch(`${API_URL}/api/admin/skills/${skill._id}`, {
        method: "DELETE",
        headers: {
          Authorization: `Bearer ${token}`,
        },
      });

      const data = await response.json();

      if (!response.ok || !data.success) {
        throw new Error(data.message || "Unable to delete skill.");
      }

      setSkills((current) => current.filter((item) => item._id !== skill._id));
      setPendingDelete(null);

      setModal({
        type: "success",
        variant: "success",
        title: "Skill Deleted!",
        message: `"${skill.name}" was deleted successfully.`,
        confirmText: "Done",
        cancelText: "Close",
      });
    } catch (err) {
      setError(err.message || "Unable to delete skill.");
      setModal(null);
      setPendingDelete(null);
    } finally {
      setDeletingId("");
    }
  }

  function closeModal() {
    if (deletingId) return;
    setModal(null);
    setPendingDelete(null);
  }

  return (
    <section className="space-y-6 text-white font-['Plus_Jakarta_Sans']">
      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4">
        <div>
          <p className="font-['DM_Mono'] text-xs uppercase tracking-[3px] text-[#FF5C00]">
            Management
          </p>
          <h2 className="text-2xl sm:text-3xl font-extrabold text-white">
            Skills Management
          </h2>
        </div>

        <div className="flex items-center gap-3 w-full sm:w-auto justify-between sm:justify-end">
          <span className="inline-flex items-center px-3 py-2 rounded-xl bg-[#FF5C00]/10 border border-[#FF5C00]/30 font-['DM_Mono'] text-xs font-bold text-[#FF5C00]">
            {skills.length} Skills
          </span>

          {!showForm && (
            <button
              type="button"
              onClick={startAdd}
              className="inline-flex items-center gap-2 rounded-xl bg-[#FF5C00] px-4 py-2.5 text-sm font-bold text-white transition hover:bg-[#cc4900] cursor-pointer"
            >
              <Plus size={18} /> Add Skill
            </button>
          )}
        </div>
      </div>

      {error && (
        <div
          role="alert"
          className="rounded-xl border border-red-500/30 bg-red-500/10 px-4 py-3 text-sm text-red-300"
        >
          {error}
          <button
            type="button"
            onClick={loadSkills}
            className="ml-3 font-semibold text-white underline"
          >
            Retry
          </button>
        </div>
      )}

      {showForm && (
        <form
          onSubmit={handleSubmit}
          className="space-y-5 rounded-2xl border border-white/10 bg-[#0a0a0a] p-5 sm:p-7"
        >
          <div className="flex items-center justify-between gap-3">
            <h3 className="text-xl font-bold text-white">
              {editingId ? "Edit Skill" : "Add New Skill"}
            </h3>

            <button
              type="button"
              onClick={() => {
                setShowForm(false);
                setEditingId(null);
                setError("");
              }}
              className="rounded-xl border border-white/10 px-4 py-2 text-xs font-semibold text-white/75 hover:bg-white/5 cursor-pointer"
            >
              Cancel
            </button>
          </div>

          <div className="grid gap-4 sm:grid-cols-2">
            <div>
              <label className="block text-xs font-semibold uppercase tracking-wider text-white/70 mb-1.5">
                Skill Name *
              </label>
              <input
                name="name"
                value={form.name}
                onChange={updateField}
                required
                maxLength={80}
                placeholder="e.g. MySQL"
                className="w-full rounded-xl border border-white/10 bg-[#111111] px-4 py-3 text-sm text-white outline-none focus:border-[#FF5C00]"
              />
            </div>

            <div>
              <label className="block text-xs font-semibold uppercase tracking-wider text-white/70 mb-1.5">
                Icon Name or Identifier *
              </label>
              <div className="flex items-center gap-3">
                <input
                  name="icon"
                  value={form.icon}
                  onChange={updateField}
                  required
                  maxLength={100}
                  placeholder="e.g. mysql or logos:docker"
                  className="flex-1 rounded-xl border border-white/10 bg-[#111111] px-4 py-3 text-sm text-white outline-none focus:border-[#FF5C00]"
                />
                <div
                  className="w-11 h-11 rounded-xl border border-white/10 bg-[#111111] flex items-center justify-center shrink-0 p-1"
                  title="Live icon preview"
                >
                  <IconPreview icon={form.icon} color={form.color} size={24} />
                </div>
              </div>
              <span className="text-[11px] text-white/40 mt-1 block">
                Type <code className="text-white/60">mysql</code>, <code className="text-white/60">python</code> or <code className="text-white/60">logos:docker</code>.
              </span>
            </div>

            <div>
              <label className="block text-xs font-semibold uppercase tracking-wider text-white/70 mb-1.5">
                Brand Color (Hex) *
              </label>
              <div className="flex gap-3 items-center">
                <input
                  type="color"
                  name="color"
                  value={form.color}
                  onChange={updateField}
                  className="w-12 h-11 rounded-xl border border-white/10 bg-[#111111] p-1 cursor-pointer"
                />
                <input
                  type="text"
                  name="color"
                  value={form.color}
                  onChange={updateField}
                  required
                  className="flex-1 rounded-xl border border-white/10 bg-[#111111] px-4 py-3 text-sm text-white outline-none focus:border-[#FF5C00]"
                  placeholder="e.g. #00758F"
                />
              </div>
            </div>

            <div>
              <label className="block text-xs font-semibold uppercase tracking-wider text-white/70 mb-1.5">
                Display Order
              </label>
              <input
                type="number"
                name="order"
                value={form.order}
                onChange={updateField}
                min="0"
                step="1"
                required
                className="w-full rounded-xl border border-white/10 bg-[#111111] px-4 py-3 text-sm text-white outline-none focus:border-[#FF5C00]"
              />
            </div>
          </div>

          <div className="flex gap-6 pt-2">
            <label className="flex items-center gap-2 text-sm cursor-pointer">
              <input
                type="checkbox"
                name="published"
                checked={form.published}
                onChange={updateField}
                className="accent-[#FF5C00] w-4 h-4"
              />
              Published on portfolio
            </label>
          </div>

          <div className="flex justify-end gap-3 pt-4 border-t border-white/10">
            <button
              type="submit"
              disabled={saving}
              className="rounded-xl bg-[#FF5C00] px-6 py-3 text-sm font-bold text-white transition hover:bg-[#cc4900] disabled:opacity-50 cursor-pointer"
            >
              {saving ? "Saving..." : editingId ? "Update Skill" : "Save Skill"}
            </button>
          </div>
        </form>
      )}

      {loading ? (
        <p className="text-white/70 text-sm">Loading skills...</p>
      ) : skills.length === 0 ? (
        <div className="rounded-2xl border border-white/10 bg-[#0a0a0a] p-10 text-center">
          <Code2 size={40} className="mx-auto text-[#FF5C00] mb-3" />
          <p className="text-white/80">No skills added yet.</p>
        </div>
      ) : (
        <div className="grid gap-4 sm:grid-cols-2 md:grid-cols-3">
          {skills.map((skill) => (
            <article
              key={skill._id}
              className="rounded-2xl border border-white/10 bg-[#0a0a0a] p-5 flex flex-col justify-between"
            >
              <div className="flex items-start justify-between gap-3">
                <div className="flex items-center gap-3 min-w-0">
                  <div className="w-10 h-10 rounded-xl border border-white/10 bg-[#111111] flex items-center justify-center shrink-0 p-1.5">
                    <IconPreview icon={skill.icon || skill.key} color={skill.color || "#FF5C00"} size={22} />
                  </div>
                  <div className="min-w-0">
                    <h3 className="font-bold text-base text-white truncate">{skill.name}</h3>
                    <p className="font-['DM_Mono'] text-[11px] text-white/40 truncate">
                      Key: {skill.icon || skill.key}
                    </p>
                  </div>
                </div>

                <span
                  className={`shrink-0 rounded-full border px-2 py-0.5 text-[10px] font-semibold ${
                    skill.published !== false
                      ? "border-emerald-400/30 bg-emerald-400/10 text-emerald-300"
                      : "border-amber-400/30 bg-amber-400/10 text-amber-300"
                  }`}
                >
                  {skill.published !== false ? "Published" : "Draft"}
                </span>
              </div>

              <div className="mt-4 flex items-center justify-between text-xs text-white/40 font-['DM_Mono']">
                <span>Order: {skill.order ?? 1}</span>
                <span>{skill.color || "#FF5C00"}</span>
              </div>

              <div className="flex gap-2 mt-4 pt-3 border-t border-white/10">
                <button
                  type="button"
                  onClick={() => startEdit(skill)}
                  className="flex-1 py-2 bg-white/5 hover:bg-white/10 rounded-lg text-xs font-semibold flex items-center justify-center gap-1.5 cursor-pointer"
                >
                  <Edit size={14} /> Edit
                </button>
                <button
                  type="button"
                  onClick={() => {
                    setPendingDelete(skill);
                    setModal({
                      type: "confirmDelete",
                      title: "Delete Skill?",
                      message: `Are you sure you want to delete "${skill.name}"?`,
                      variant: "danger",
                      confirmText: "Delete",
                      cancelText: "Cancel",
                    });
                  }}
                  disabled={deletingId === skill._id}
                  className="flex-1 py-2 bg-red-500/10 hover:bg-red-500/20 text-red-300 rounded-lg text-xs font-semibold flex items-center justify-center gap-1.5 cursor-pointer disabled:opacity-50"
                >
                  <Trash2 size={14} /> {deletingId === skill._id ? "Deleting..." : "Delete"}
                </button>
              </div>
            </article>
          ))}
        </div>
      )}

      <ConfirmationModal
        isOpen={Boolean(modal)}
        title={modal?.title || ""}
        message={modal?.message || ""}
        variant={modal?.variant || "danger"}
        confirmText={modal?.confirmText || "Confirm"}
        cancelText={modal?.cancelText || "Close"}
        isLoading={Boolean(deletingId)}
        onConfirm={() => {
          if (modal?.type === "success") {
            setModal(null);
            return;
          }
          confirmDelete();
        }}
        onCancel={closeModal}
      />
    </section>
  );
}

SkillManagement.propTypes = {
  token: PropTypes.string.isRequired,
};

function OrangeCanvas() {
  const canvasRef = useRef(null);
  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    let animId;
    const particles = [];
    const resize = () => {
      canvas.width = canvas.offsetWidth;
      canvas.height = canvas.offsetHeight;
    };
    resize();
    window.addEventListener("resize", resize);
    for (let i = 0; i < 60; i++) {
      particles.push({
        x: Math.random() * canvas.width,
        y: Math.random() * canvas.height,
        r: Math.random() * 2 + 0.5,
        vx: (Math.random() - 0.5) * 0.4,
        vy: (Math.random() - 0.5) * 0.4,
        alpha: Math.random() * 0.6 + 0.1,
      });
    }
    const draw = () => {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      for (let i = 0; i < particles.length; i++) {
        for (let j = i + 1; j < particles.length; j++) {
          const dx = particles[i].x - particles[j].x,
            dy = particles[i].y - particles[j].y;
          const dist = Math.sqrt(dx * dx + dy * dy);
          if (dist < 100) {
            ctx.beginPath();
            ctx.strokeStyle = `rgba(255,92,0,${(1 - dist / 100) * 0.12})`;
            ctx.lineWidth = 0.8;
            ctx.moveTo(particles[i].x, particles[i].y);
            ctx.lineTo(particles[j].x, particles[j].y);
            ctx.stroke();
          }
        }
      }
      particles.forEach((p) => {
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
        ctx.fillStyle = `rgba(255,92,0,${p.alpha})`;
        ctx.fill();
        p.x += p.vx;
        p.y += p.vy;
        if (p.x < 0 || p.x > canvas.width) p.vx *= -1;
        if (p.y < 0 || p.y > canvas.height) p.vy *= -1;
      });
      animId = requestAnimationFrame(draw);
    };
    draw();
    return () => {
      cancelAnimationFrame(animId);
      window.removeEventListener("resize", resize);
    };
  }, []);
  return <canvas ref={canvasRef} className="absolute inset-0 w-full h-full" />;
}

function normalizeIconIdentifier(icon) {
  const raw = String(icon || "").trim();
  if (!raw) return "";
  if (LEGACY_ICONS[raw]) return LEGACY_ICONS[raw];
  if (raw.includes(":")) return raw.toLowerCase();
  const slug = raw.toLowerCase().replace(/\s+/g, "-");
  return `simple-icons:${SIMPLE_ICON_ALIASES[slug] || slug}`;
}

function getIconUrl(icon) {
  const id = normalizeIconIdentifier(icon);
  const match = id.match(/^([a-z0-9-]+):([a-z0-9-]+)$/i);
  if (!match) return null;
  return `https://api.iconify.design/${match[1]}/${match[2]}.svg`;
}

function SkillIcon({ icon, color, name }) {
  const src = getIconUrl(icon);
  const [failedSrc, setFailedSrc] = useState("");
  const failed = Boolean(src && failedSrc === src);

  if (!src || failed) {
    return (
      <span
        className="inline-flex shrink-0 items-center justify-center rounded-lg bg-white/5 font-bold text-slate-400"
        style={{ width: 32, height: 32 }}
      >
        {String(icon || "?").trim().slice(0, 1).toUpperCase() || "?"}
      </span>
    );
  }

  return (
    <span
      key={src}
      className="inline-block shrink-0"
      role="img"
      aria-label={`${name} icon`}
      style={{
        width: 32,
        height: 32,
        backgroundColor: color || "#FF5C00",
        WebkitMask: `url("${src}") center / contain no-repeat`,
        mask: `url("${src}") center / contain no-repeat`,
      }}
    >
      <img src={src} alt="" width={32} height={32} onError={() => setFailedSrc(src)} style={{ display: "none" }} />
    </span>
  );
}
